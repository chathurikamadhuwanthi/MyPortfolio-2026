/* ---- longest prefix match ---- */
struct route {
    uint32_t prefix;
    uint32_t mask;
    uint8_t  plen;
    uint32_t next_hop;
    char     iface[16];
};

static struct route table[MAX_ROUTES];
static int          nroutes;

const struct route *lookup(uint32_t dst)
{
    const struct route *best = NULL;

    for (int i = 0; i < nroutes; i++) {
        const struct route *r = &table[i];

        if ((dst & r->mask) != r->prefix)
            continue;
        if (!best || r->plen > best->plen)
            best = r;
    }
    return best;
}

void add_route(const char *cidr, const char *via,
               const char *dev)
{
    struct route *r = &table[nroutes++];
    char addr[32];
    int  plen;

    sscanf(cidr, "%31[^/]/%d", addr, &plen);
    r->plen   = plen;
    r->mask   = plen ? ~0u << (32 - plen) : 0;
    r->prefix = ntohl(inet_addr(addr)) & r->mask;
    r->next_hop = ntohl(inet_addr(via));
    strncpy(r->iface, dev, sizeof r->iface - 1);
}

/* ---- 802.1Q + ARP ---- */
#define ETH_P_8021Q 0x8100
#define ETH_P_ARP   0x0806

struct vlan_tag {
    uint16_t tci;        /* pcp:3 dei:1 vid:12 */
    uint16_t ethertype;
};

static inline uint16_t vlan_id(const struct vlan_tag *t)
{
    return ntohs(t->tci) & 0x0fff;
}

void handle_frame(const uint8_t *f, size_t len)
{
    const struct eth_hdr *eth = (const void *)f;
    uint16_t type = ntohs(eth->ethertype);
    size_t   off  = sizeof *eth;

    if (type == ETH_P_8021Q) {
        const struct vlan_tag *t = (const void *)(f + off);
        log_vlan(vlan_id(t));
        type = ntohs(t->ethertype);
        off += sizeof *t;
    }

    switch (type) {
    case ETH_P_ARP:
        arp_input(f + off, len - off);
        break;
    case ETH_P_IP:
        ip_input(f + off, len - off);
        break;
    default:
        stats.dropped++;
    }
}

for (int i = 0; i < nports; i++) {
    if (port[i].state == FORWARDING)
        flood(&port[i], frame, len);
}

// ---- checksum + ICMP echo ----
uint16_t checksum(const void *data, size_t len)
{
    const uint16_t *p = data;
    uint32_t sum = 0;

    while (len > 1) {
        sum += *p++;
        len -= 2;
    }
    if (len)
        sum += *(const uint8_t *)p;

    while (sum >> 16)
        sum = (sum & 0xffff) + (sum >> 16);

    return (uint16_t)~sum;
}

int send_echo(int fd, struct in_addr dst, uint16_t seq)
{
    struct icmphdr icmp = {0};
    icmp.type             = ICMP_ECHO;
    icmp.code             = 0;
    icmp.un.echo.id       = htons(getpid() & 0xffff);
    icmp.un.echo.sequence = htons(seq);
    icmp.checksum         = checksum(&icmp, sizeof icmp);

    struct sockaddr_in to = {
        .sin_family = AF_INET,
        .sin_addr   = dst,
    };
    return sendto(fd, &icmp, sizeof icmp, 0,
                  (struct sockaddr *)&to, sizeof to);
}

/* ttl exceeded -> hop reached */
if (icmp->type == ICMP_TIME_EXCEEDED) {
    printf("%2d  %s  %.2f ms\n", ttl,
           inet_ntoa(from.sin_addr), rtt_ms);
    continue;
}

/* ---- 802.1Q + ARP ---- */
#define ETH_P_8021Q 0x8100
#define ETH_P_ARP   0x0806

struct vlan_tag {
    uint16_t tci;        /* pcp:3 dei:1 vid:12 */
    uint16_t ethertype;
};

static inline uint16_t vlan_id(const struct vlan_tag *t)
{
    return ntohs(t->tci) & 0x0fff;
}

void handle_frame(const uint8_t *f, size_t len)
{
    const struct eth_hdr *eth = (const void *)f;
    uint16_t type = ntohs(eth->ethertype);
    size_t   off  = sizeof *eth;

    if (type == ETH_P_8021Q) {
        const struct vlan_tag *t = (const void *)(f + off);
        log_vlan(vlan_id(t));
        type = ntohs(t->ethertype);
        off += sizeof *t;
    }

    switch (type) {
    case ETH_P_ARP:
        arp_input(f + off, len - off);
        break;
    case ETH_P_IP:
        ip_input(f + off, len - off);
        break;
    default:
        stats.dropped++;
    }
}

for (int i = 0; i < nports; i++) {
    if (port[i].state == FORWARDING)
        flood(&port[i], frame, len);
}
